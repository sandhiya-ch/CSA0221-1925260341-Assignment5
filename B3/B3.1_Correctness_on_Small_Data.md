B3.1 – Correctness on Small Data

1. Objective

To write and read 10 donor records using both text and binary files and verify that the data is identical.

2. C Program

#include <stdio.h>
#include <string.h>

typedef struct {
    int id;
    char name[30];
    char bloodGroup[5];
    int age;
} Donor;

int main() {
    Donor donors[10], textData[10], binaryData[10];
    FILE *fp;
    
    for (int i = 0; i < 10; i++) {
        donors[i].id = i + 1;
        sprintf(donors[i].name, "Donor%d", i + 1);
        strcpy(donors[i].bloodGroup, "O+");
        donors[i].age = 20 + i;
    }

    /* Write text file */
    fp = fopen("donors.txt", "w");
    if (fp == NULL) {
        perror("Text file");
        return 1;
    }

    for (int i = 0; i < 10; i++) {
        fprintf(fp, "%d %s %s %d\n",
                donors[i].id, donors[i].name,
                donors[i].bloodGroup, donors[i].age);
    }
    fclose(fp);

    /* Read text file */
    fp = fopen("donors.txt", "r");
    if (fp == NULL) {
        perror("Text file");
        return 1;
    }

    for (int i = 0; i < 10; i++) {
        fscanf(fp, "%d %29s %4s %d",
               &textData[i].id, textData[i].name,
               textData[i].bloodGroup, &textData[i].age);
    }
    fclose(fp);

    /* Write binary file */
    fp = fopen("donors.dat", "wb");
    if (fp == NULL) {
        perror("Binary file");
        return 1;
    }

    fwrite(donors, sizeof(Donor), 10, fp);
    fclose(fp);

    /* Read binary file */
    fp = fopen("donors.dat", "rb");
    if (fp == NULL) {
        perror("Binary file");
        return 1;
    }

    fread(binaryData, sizeof(Donor), 10, fp);
    fclose(fp);

    /* Compare records */
    int textMatch = 1, binaryMatch = 1;

    for (int i = 0; i < 10; i++) {
        if (donors[i].id != textData[i].id ||
            strcmp(donors[i].name, textData[i].name) != 0 ||
            strcmp(donors[i].bloodGroup, textData[i].bloodGroup) != 0 ||
            donors[i].age != textData[i].age)
            textMatch = 0;

        if (donors[i].id != binaryData[i].id ||
            strcmp(donors[i].name, binaryData[i].name) != 0 ||
            strcmp(donors[i].bloodGroup, binaryData[i].bloodGroup) != 0 ||
            donors[i].age != binaryData[i].age)
            binaryMatch = 0;
    }

    printf("Text file verification: %s\n",
           textMatch ? "PASS" : "FAIL");
    printf("Binary file verification: %s\n",
           binaryMatch ? "PASS" : "FAIL");

    return 0;
}

3. Expected Output

Text file verification: PASS
Binary file verification: PASS

4. Observation

- Ten donor records were written into a text file.
- The same records were written into a binary file.
- Both files were read back successfully.
- The original and read-back records were compared.

5. Result

All 10 donor records were read correctly from both text and binary files. The verification was successful.
